# The engine on its own, and he_render for test/parity.js
cmake_minimum_required(VERSION 3.22)
project(he_parity CXX)
set(CMAKE_CXX_STANDARD 17)
add_library(he_core STATIC ../core/he_core.cpp)
target_include_directories(he_core PUBLIC ../core)
if (NOT MSVC)
  target_compile_options(he_core PRIVATE -O2 -fno-fast-math -ffp-contract=off)
else()
  target_compile_options(he_core PRIVATE /fp:precise)
endif()
add_executable(he_render ../tools/he_render.cpp)
target_link_libraries(he_render he_core)
# Sync to host with a fake playhead
add_executable(sync_test ../test/sync_test.cpp)
target_link_libraries(sync_test he_core)
# A CC must not flicker back to the host's stale parameter value (src/ParamHold.h)
add_executable(param_hold_test ../test/param_hold_test.cpp)
# Every preset from 1 kHz to 768 kHz renders finite audio (clap-validator process-varying-sample-rates)
add_executable(samplerate_test ../test/samplerate_test.cpp)
target_link_libraries(samplerate_test he_core)
# GOODBYE is a hard stop: silent within 50 ms, no click, nothing restarts (1.0.3)
add_executable(goodbye_test ../test/goodbye_test.cpp)
target_link_libraries(goodbye_test he_core)
# A drag pulls as smoothly at any host buffer size: gliding moves (1.0.4)
add_executable(drag_test ../test/drag_test.cpp)
target_link_libraries(drag_test he_core)
